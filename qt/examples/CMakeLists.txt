cmake_minimum_required(VERSION 3.24)
project(RadarBinarySdkConsumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets)
if(NOT RADAR_SDK)
  message(FATAL_ERROR "Set RADAR_SDK to delivered sdk directory")
endif()
add_executable(radar-sdk-consumer binary_consumer.cpp)
target_include_directories(radar-sdk-consumer PRIVATE "${RADAR_SDK}/include")
target_link_libraries(radar-sdk-consumer PRIVATE "${RADAR_SDK}/lib/radar_contracts.lib" Qt6::Widgets)
if(MSVC)
  target_compile_options(radar-sdk-consumer PRIVATE /utf-8)
endif()
