cmake_minimum_required(VERSION 3.24)
project(RadarQtPlayback VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
foreach(config RELEASE DEBUG RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
  set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
  set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/lib")
endforeach()
if(MSVC)
  add_compile_options(/utf-8 /W4 /permissive-)
endif()
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Test)
add_library(radar_contracts SHARED contracts/contracts.cpp include/radar/contracts.h)
target_compile_definitions(radar_contracts PRIVATE RADAR_CONTRACTS_BUILD)
target_include_directories(radar_contracts PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(radar_contracts PUBLIC Qt6::Core Qt6::Gui Qt6::Widgets)
foreach(module data_reader playback frontend)
  add_subdirectory(modules/${module})
endforeach()
add_executable(radar-playback apps/main.cpp)
target_link_libraries(radar-playback PRIVATE radar_contracts radar_data_reader radar_playback radar_frontend Qt6::Widgets)
include(CTest)
if(BUILD_TESTING)
  foreach(module contracts data_reader playback frontend)
    add_executable(${module}_test tests/${module}_test.cpp)
    target_link_libraries(${module}_test PRIVATE radar_contracts Qt6::Test)
    if(NOT module STREQUAL "contracts")
      target_link_libraries(${module}_test PRIVATE radar_${module})
    endif()
    add_test(NAME ${module} COMMAND ${module}_test)
    set_tests_properties(${module} PROPERTIES TIMEOUT 60 ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QPA_FONTDIR=$ENV{SystemRoot}/Fonts")
  endforeach()
endif()
install(TARGETS radar-playback radar_contracts radar_data_reader radar_playback radar_frontend
  RUNTIME DESTINATION bin LIBRARY DESTINATION bin ARCHIVE DESTINATION sdk/lib)
install(FILES include/radar/contracts.h DESTINATION sdk/include/radar)
install(DIRECTORY docs examples DESTINATION .)
